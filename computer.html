1.#WAP in C that stores ID , fullname , post and Salary of 10 employee in a structure and display record of only those employees whose Salary is greater than 5000 and less then 7500

Structure 
#include <stdio.h>

struct Employee {
    int id;
    char fullname[50];
    char post[30];
    float salary;
};

int main() {
    struct Employee e[10];
    int i;

    // Input details of 10 employees
    for (i = 0; i < 10; i++) {
        printf("\nEnter details of Employee %d:\n", i + 1);

        printf("ID: ");
        scanf("%d", &e[i].id);

        printf("Full Name: ");
        scanf(" %[^\n]", e[i].fullname);

        printf("Post: ");
        scanf(" %[^\n]", e[i].post);

        printf("Salary: ");
        scanf("%f", &e[i].salary);
    }

    // Display employees with salary > 5000 and < 7500
    printf("\n========== EMPLOYEE RECORD ==========\n");

    for (i = 0; i < 10; i++) {
        if (e[i].salary > 5000 && e[i].salary < 7500) {
            printf("\nID: %d", e[i].id);
            printf("\nFull Name: %s", e[i].fullname);
            printf("\nPost: %s", e[i].post);
            printf("\nSalary: %.2f\n", e[i].salary);
        }
    }

    return 0;
}

3.WAP in C that stores name and marks Secured by 10 Students in math, physics and chemistry in a structure and displays record of all Students along with obtained marks and percentage and also remarks Showing student is pass or fail. Assume 40 pass mark for each subject.

Structure 
 #include <stdio.h>

struct Student {
    char name[50];
    float math, physics, chemistry;
    float total, percentage;
};

int main() {
    struct Student s[10];
    int i;

    // Input details
    for (i = 0; i < 10; i++) {
        printf("\nEnter details of Student %d:\n", i + 1);

        printf("Name: ");
        scanf(" %[^\n]", s[i].name);

        printf("Math marks: ");
        scanf("%f", &s[i].math);

        printf("Physics marks: ");
        scanf("%f", &s[i].physics);

        printf("Chemistry marks: ");
        scanf("%f", &s[i].chemistry);

        s[i].total = s[i].math + s[i].physics + s[i].chemistry;
        s[i].percentage = s[i].total / 3;
    }

    // Display records
    printf("\n========== STUDENT RECORD ==========\n");

    for (i = 0; i < 10; i++) {
        printf("\nName: %s", s[i].name);
        printf("\nMath: %.2f", s[i].math);
        printf("\nPhysics: %.2f", s[i].physics);
        printf("\nChemistry: %.2f", s[i].chemistry);
        printf("\nTotal: %.2f", s[i].total);
        printf("\nPercentage: %.2f%%", s[i].percentage);

        if (s[i].math >= 40 &&
            s[i].physics >= 40 &&
            s[i].chemistry >= 40)
            printf("\nRemarks: PASS\n");
        else
            printf("\nRemarks: FAIL\n");
    }

    return 0;
}
